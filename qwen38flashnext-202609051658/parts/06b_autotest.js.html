/* ---------- a shallow greedy player used by the harnesses --------------- */
const fullRowsOf = g => {
  const out = [];
  for (let y = 0; y < HEIGHT; y++) {
    let f = true;
    for (let x = 0; x < COLS; x++) if (!g[idx(x, y)]) { f = false; break; }
    if (f) out.push(y);
  }
  return out;
};
const collapseOf = (g, rows) => {
  const set = {}; rows.forEach(y => set[y] = 1);
  const out = new Array(HEIGHT * COLS).fill(null);
  let wy = HEIGHT - 1;
  for (let y = HEIGHT - 1; y >= 0; y--) {
    if (set[y]) continue;
    for (let x = 0; x < COLS; x++) out[idx(x, wy)] = g[idx(x, y)];
    wy--;
  }
  return out;
};
const scoreBoard = g => {
  const lines = fullRowsOf(g).length;
  const h = [];
  let holes = 0, agg = 0, bump = 0, well = 0, maxH = 0;
  for (let x = 0; x < COLS; x++) {
    let seen = false, colHoles = 0, top = -1;
    for (let y = 0; y < HEIGHT; y++) {
      if (g[idx(x, y)]) { if (!seen) { seen = true; top = y; } }
      else if (seen) colHoles++;
    }
    h.push(seen ? HEIGHT - top : 0);
    holes += colHoles;
  }
  for (let x = 0; x < COLS; x++) { agg += h[x]; if (h[x] > maxH) maxH = h[x]; }
  for (let x = 0; x + 1 < COLS; x++) bump += Math.abs(h[x] - h[x + 1]);
  for (let x = 0; x < COLS; x++) {
    const l = x ? h[x - 1] : 1e3, r = x < COLS - 1 ? h[x + 1] : 1e3;
    const d = Math.min(l, r) - h[x];
    if (d > 0) well += Math.min(d, 4);
  }
  return lines * 4 - holes * 4 - agg * .2 - bump * .8 - maxH * .2 + well * .3;
};
const bestMove = () => {
  const t = S.piece.t;
  let best = null;
  for (let r = 0; r < (t === 'O' ? 1 : 4); r++) {
    for (let x = -2; x <= COLS + 1; x++) {
      if (!fits(S.grid, t, r, x, 0)) continue;
      let y = 0, guard = 0;
      while (fits(S.grid, t, r, x, y + 1) && guard++ < 40) y++;
      if (!fits(S.grid, t, r, x, y)) continue;
      const g = S.grid.slice();
      for (const [cx, cy] of SHAPES[t][r]) {
        const X = x + cx, Y = y + cy;
        if (Y >= 0 && Y < HEIGHT && X >= 0 && X < COLS) g[idx(X, Y)] = t;
      }
      const rows = fullRowsOf(g);
      const sc = scoreBoard(rows.length ? collapseOf(g, rows) : g);
      if (!best || sc > best.sc) best = { r, x, y, sc };
    }
  }
  return best;
};
const botPiece = () => {
  if (S.phase !== 'playing' || !S.piece) return false;
  const m = bestMove();
  if (!m) return false;
  for (let i = 0; i < 4 && S.piece.r !== m.r; i++) {
    const step = (m.r - S.piece.r + 4) % 4 <= 2 ? 1 : -1;
    if (rotate(S, step) === false) break;
  }
  let guard = 0;
  while (S.piece && S.piece.x !== m.x && guard++ < 20) if (!move(S, S.piece.x < m.x ? 1 : -1)) break;
  action('hard');
  return true;
};

/* run frames until the state machine is idle again */
const settleFrames = (max = 300) => {
  for (let i = 0; i < max && (S.phase !== 'playing' || S.clearing); i++) { update(1 / 60); drawFrame(); }
};

/* ---------- #autotest --------------------------------------------------- */
if (location.hash.indexOf('autotest') >= 0) {
  const rep = { ok: false, errors: [], warns: {}, actions: {}, frames: 0 };
  const note = k => { rep.actions[k] = (rep.actions[k] || 0) + 1; };
  const log = (...a) => { rep.errors.push(a.map(String).join(' ')); };
  window.addEventListener('error', e => log('window.onerror: ' + (e.message || e)));
  const oldErr = console.error, oldWarn = console.warn;
  console.error = (...a) => { log('console.error: ' + a.join(' ')); oldErr.apply(console, a); };
  console.warn = (...a) => { const k = a.map(String).join(' '); rep.warns[k] = (rep.warns[k] || 0) + 1; };

  const run = () => {
    try {
      startGame();
      const clearSizes = [];
      let lastLines = 0, lastReported = 0, totalPieces = 0, over = 0, peak = 0;
      const themes = {};
      const LIMIT = 60 * 900;
      for (let f = 0; f < LIMIT && totalPieces < 900; f++) {
        if (botPiece()) note('piece');
        update(1 / 60);
        drawFrame();
        rep.frames++;
        themes[FX.theme] = (themes[FX.theme] || 0) + 1;
        totalPieces += Math.max(0, S.stats.pieces - lastReported); lastReported = S.stats.pieces;
        peak = Math.max(peak, S.lines);
        if (S.lines !== lastLines) { clearSizes.push(S.lines - lastLines); lastLines = S.lines; }
        if (appPhase === 'over') { over++; startGame(); lastReported = 0; lastLines = 0; }
      }
      pauseGame(true); update(1 / 60); drawFrame();
      pauseGame(false); update(1 / 60); drawFrame();

      settleFrames();
      /* drill 1 — perfect clear: two rows missing exactly the O footprint */
      const perfectBefore = S.stats.perfect;
      S.grid = new Array(HEIGHT * COLS).fill(null);
      for (const y of [HEIGHT - 2, HEIGHT - 1])
        for (const x of [0, 1, 2, 3, 6, 7, 8, 9]) S.grid[idx(x, y)] = TYPES[x % 7];
      spawn(S, 'O');
      action('hard');
      for (let f = 0; f < 120; f++) { update(1 / 60); drawFrame(); }
      rep.perfectDrill = S.stats.perfect - perfectBefore;

      settleFrames();
      /* drill 2 — T-spin double: only the last wall-kick test fits */
      const spinBefore = S.stats.tspin;
      S.grid = new Array(HEIGHT * COLS).fill(null);
      for (const x of [0, 1, 2, 3, 5, 6, 7, 8, 9]) S.grid[idx(x, HEIGHT - 1)] = 'L';
      for (const x of [0, 1, 2, 3, 6, 7, 8, 9]) S.grid[idx(x, HEIGHT - 2)] = 'J';
      for (const x of [0, 1, 2, 3, 5, 6, 8, 9]) S.grid[idx(x, HEIGHT - 3)] = 'S';
      S.grid[idx(4, HEIGHT - 5)] = 'I';      /* blocks the first two kick tests */
      spawn(S, 'T');
      S.piece.x = 4; S.piece.y = HEIGHT - 5; /* exactly the unit-test setup */
      rep.spinKick = rotate(S, 1);
      action('hard');
      for (let f = 0; f < 120; f++) { update(1 / 60); drawFrame(); }
      rep.tspinDrill = S.stats.tspin - spinBefore;

      action('hold'); update(1 / 60); drawFrame();
      endGame('autotest');
      for (let f = 0; f < 90; f++) { update(1 / 60); drawFrame(); }

      rep.over = over;
      rep.score = S.score; rep.lines = S.lines; rep.level = S.level; rep.peakLines = peak;
      rep.stats = S.stats;
      rep.themeTime = themes;
      rep.clears = {
        single: clearSizes.filter(n => n === 1).length,
        double: clearSizes.filter(n => n === 2).length,
        triple: clearSizes.filter(n => n === 3).length,
        tetris: clearSizes.filter(n => n >= 4).length
      };
      rep.maxParticles = FX.count;
      rep.totalPieces = totalPieces;
      rep.best = best;
      rep.ok = rep.errors.length === 0 && totalPieces > 800 && peak > 300 &&
        Object.keys(themes).length >= 6 && rep.perfectDrill >= 1 && rep.tspinDrill >= 1;
    } catch (e) {
      log('EXCEPTION: ' + (e && e.stack ? e.stack : e));
    }
    document.title = 'AUTOTEST ' + (rep.ok ? 'PASS' : 'FAIL');
    const pre = document.createElement('pre');
    pre.id = 'autotest';
    pre.style.cssText = 'position:fixed;left:0;top:0;z-index:99;font:11px monospace;background:#000;color:#8fff;padding:8px;max-height:100%;overflow:auto';
    pre.textContent = JSON.stringify(rep, null, 1);
    document.body.appendChild(pre);
  };
  if (document.readyState === 'complete' || document.readyState === 'interactive') setTimeout(run, 60);
  else window.addEventListener('DOMContentLoaded', () => setTimeout(run, 60));
